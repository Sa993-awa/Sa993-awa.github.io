import unittest

class TestAuthenticationSystem(unittest.TestCase):

    def setUp(self):
        self.auth_system = AuthenticationSystem()

    def test_add_user_successfully(self):
        result = self.auth_system.add_user( "Sally", "Sally1993@","Sally@uniuk.co.uk")
        self.assertTrue(result)

    def test_reject_empty_username(self):
        result = self.auth_system.add_user("","Sally1993@","Sally@uniuk.co.uk")
        self.assertFalse(result)

    def test_reject_empty_password(self):
        result = self.auth_system.add_user("Sally","","Sally@uniuk.co.uk")
        self.assertFalse(result)

    def test_reject_short_password(self):
        result = self.auth_system.add_user( "Sally", "Sally123@","Sally@uniuk.co.uk")

        self.assertFalse(result)

    def test_reject_password_without_lowercase(self):
        result = self.auth_system.add_user( "Sally", "SALLY12345@","Sally@uniuk.co.uk")

        self.assertFalse(result)

    def test_reject_password_without_uppercase(self):
        result = self.auth_system.add_user("Sally","sally12345@","Sally@uniuk.co.uk")

        self.assertFalse(result)

    def test_reject_password_without_symbol(self):
        result = self.auth_system.add_user( "Sally", "Sally123456","Sally@uniuk.co.uk")

        self.assertFalse(result)

    def test_reject_duplicate_username(self):
        self.auth_system.add_user("Sally", "Sally1993@","Sally@uniuk.co.uk" )
        result = self.auth_system.add_user( "Sally", "Another123@","Sally@uniuk.co.uk")
        self.assertFalse(result)

    def test_successful_authentication(self):
        self.auth_system.add_user("Sally","Sally1993@")
        result = self.auth_system.authenticate("Sally","Sally1993@")
        self.assertTrue(result)

    def test_failed_authentication(self):
        self.auth_system.add_user( "Sally","Sally1993@","Sally@uniuk.co.uk")
        result = self.auth_system.authenticate("Sally","WrongPassword123@","Sally@uniuk.co.uk")
        self.assertFalse(result)

    def test_three_failed_attempts_are_allowed(self):
        self.auth_system.add_user("Sally","Sally1993@","Sally@uniuk.co.uk")

        for _ in range(3):
            result = self.auth_system.authenticate("Sally", "WrongPassword123@","Sally@uniuk.co.uk")

            self.assertFalse(result)

    def test_fourth_attempt_is_blocked(self):
        self.auth_system.add_user("Sally","Sally1993@","Sally@uniuk.co.uk")

        for _ in range(3):
            self.auth_system.authenticate("Sally","WrongPassword123@","Sally@uniuk.co.uk" )

        result = self.auth_system.authenticate("Sally","Sally1993@","Sally@uniuk.co.uk")

        self.assertFalse(result)

    def test_successful_login_resets_failed_attempts(self):
        self.auth_system.add_user( "Sally", "Sally1993@","Sally@uniuk.co.uk")

        # Two failed attempts
        self.auth_system.authenticate( "Sally", "WrongPassword123@","Sally@uniuk.co.uk" )

        self.auth_system.authenticate( "Sally","WrongPassword123@","Sally@uniuk.co.uk")

        # Successful login resets the counter
        result = self.auth_system.authenticate( "Sally", "Sally1993@","Sally@uniuk.co.uk")

        self.assertTrue(result)

        # Another failed attempt should now be counted from zero
        result = self.auth_system.authenticate( "Sally","WrongPassword123@","Sally@uniuk.co.uk")

        self.assertFalse(result)

    def test_password_is_hashed(self):
        self.auth_system.add_user("Sally","Sally1993@","Sally@uniuk.co.uk" )

        user = self.auth_system.get_user("Sally")

        # The stored password should not equal the original password
        self.assertNotEqual(user._password,b"Sally1993@")

        # bcrypt hashes normally start with b"$2"
        self.assertTrue( user._password.startswith(b"$2")
