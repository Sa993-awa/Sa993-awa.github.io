        self.assertFalse(result)

            self.assertFalse(result)

        self.auth_system.add_user("Sally","Sally1993@","Sally@uniuk.co.uk")

        for _ in range(3):

        result = self.auth_system.authenticate("Sally","Sally1993@","Sally@uniuk.co.uk")

        self.assertFalse(result)

    def test_successful_login_resets_failed_attempts(self):
        self.auth_system.add_user( "Sally", "Sally1993@","Sally@uniuk.co.uk")

        # Two failed attempts
        self.auth_system.authenticate( "Sally", "WrongPassword123@","Sally@uniuk.co.uk" )

        self.auth_system.authenticate( "Sally","WrongPassword123@","Sally@uniuk.co.uk")

        # Successful login resets the counter
        result = self.auth_system.authenticate( "Sally", "Sally1993@","Sally@uniuk.co.uk")

        self.assertTrue(result)

        # Another failed attempt should now be counted from zero
        result = self.auth_system.authenticate( "Sally","WrongPassword123@","Sally@uniuk.co.uk")

        self.assertFalse(result)

        self.auth_system.add_user("Sally","Sally1993@","Sally@uniuk.co.uk" )

        self.assertNotEqual(user._password,b"Sally1993@")

        self.assertTrue( user._password.startswith(b"$2")

import bcrypt


class UserManagment:
    def __init__(self, student_id, password,student_email):
        self.student_id = student_id
        self._password = bcrypt.hashpw(password.encode(),bcrypt.gensalt())
        self._student_email= student_email

class AuthenticationSystem:
    def __init__(self):
        self._users = []
        self._failed_attempts = {}

    def add_user(self, student_id, password,student_email):

        # Check for empty username or password
        if not student_id or not password or not student_email:
            return False

        # Check password length
        if len(password) < 10:
            return False

        # Check password contains lowercase, uppercase and symbol
        has_lowercase = any(char.islower() for char in password)
        has_uppercase = any(char.isupper() for char in password)
        has_symbol = any(not char.isalnum() for char in password)

        if not (has_lowercase and has_uppercase and has_symbol):
            return False

        # Prevent duplicate usernames
        for user in self._users:
            if user.student_id == student_id:
                return False

        # Store user with securely hashed password
        self._users.append(UserManagment(student_id, password,student_email))

        # Set failed login attempts to zero
        self._failed_attempts[student_id] = 0

        return True

    def authenticate(self, student_id, password):

        # Get number of previous failed attempts
        attempts = self._failed_attempts.get(student_id, 0)

        # Maximum of 3 failed attempts
        if attempts >= 3:
            return False

        # Search for the username
        for user in self._users:

            if user.student_id == student_id:

                # Compare entered password with bcrypt hash
                if bcrypt.checkpw( password.encode(), user._password ):
                    # Reset failed attempts after successful login
                    self._failed_attempts[student_id] = 0
                    return True

                else:
                    # Increase failed attempts
                    self._failed_attempts[student_id] = attempts + 1
                    return False

        return False

    def get_user(self, student_id):

        for user in self._users:
            if user.student_id == student_id:
                return user

        return None
