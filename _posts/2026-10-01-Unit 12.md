#Cache abstaction

class Cache(ABC):

    @abstractmethod
    def get(self, key):
        pass

    @abstractmethod
    def set(self, key, value):
        pass


class InMemoryCache(Cache):

    def __init__(self):
        self.__cache = {}

    def get(self, key):
        return self.__cache.get(key)

    def set(self, key, value):
        self.__cache[key] = value


# User hierarchy

class User(ABC):

    def __init__(self, user_id, name, password):

        if not user_id:
            raise ValueError("User ID is required")

        if not name:
            raise ValueError("Name is required")

        if len(password) < 8:
            raise ValueError(
                "Password must contain at least 8 characters"
            )

        self.__user_id = user_id
        self.__name = name

        # Secure password storage using bcrypt
        self.__password_hash = bcrypt.hashpw(password.encode(),bcrypt.gensalt())

    def get_user_id(self):
        return self.__user_id

    def get_name(self):
        return self.__name

    def check_password(self, password):
        return bcrypt.checkpw(password.encode(),self.__password_hash)

    @abstractmethod
    def get_role(self):
        pass

    def __str__(self):
        return f"{self.__name} ({self.get_role()})"


class Customer(User):

    def __init__(self, user_id, name, password):

        super().__init__(user_id, name, password)

        self.__cart = []

    def get_role(self):
        return "Customer"

    def add_to_cart(self, product, quantity=1):

        if quantity <= 0:
            return False

        self.__cart.append({"product": product,"quantity": quantity })

        return True

    def get_cart(self):
        return self.__cart

    def view_cart(self):

        return [f"{item['product']} x{item['quantity']}"for item in self.__cart]

    def calculate_cart_total(self):

        total = 0

        for item in self.__cart:

            total += (item["product"].get_price()* item["quantity"])

        return total

class ProductService:

    def __init__(self, product_repository, cache):

        # Two dependencies are injected.
        self.__repository = product_repository
        self.__cache = cache

    def get_product(self, product_id):

        # Check cache first
        product = self.__cache.get(product_id)

        if product is not None:
            return product

        # If not cached, use repository
        product = self.__repository.find_by_id( product_id)

        if product is not None:
            self.__cache.set(product_id,product)

        return product

    def get_all_products(self):

        return self.__repository.find_all()

 # Inventory

class Inventory:

    def __init__(self):

        self.__products = {}

    def add_product(self,product,quantity=1):

        if quantity <= 0:
            raise ValueError("Quantity must be greater than zero" )

        product_id = product.get_product_id()

        if product_id in self.__products:

            self.__products[product_id ]["quantity"] += quantity

        else:

            self.__products[product_id] = {"product": product, "quantity": quantity }

    def check_stock(self, product_id):

        item = self.__products.get( product_id )

        if item:
            return item["quantity"]

        return 0

    def reduce_stock(self,product_id, quantity):

        if quantity <= 0:
            return False

        if self.check_stock( product_id) >= quantity:
            self.__products[product_id]["quantity"] -= quantity

            return True

        return False

#Order

class Order:

    def __init__(self, order_id, customer, payment_strategy):

        self.__order_id = order_id
        self.__customer = customer

        # Strategy is also injected.
        self.__payment_strategy = payment_strategy

        self.__status = "Pending"

    def get_order_id(self):
        return self.__order_id

    def get_customer(self):
        return self.__customer

    def get_status(self):
        return self.__status

    def checkout(self, total):

        if total <= 0:

            self.__status = "Payment Failed"

            return False

        if self.__payment_strategy.pay(total):

            self.__status = "Confirmed"

            return True

        self.__status = "Payment Failed"

        return False

# Decorator Pattern

class OrderPriceComponent(ABC):

    @abstractmethod
    def calculate_price(self):
        pass


class BasicOrderPrice(OrderPriceComponent):

    def __init__(self, price):

        self.__price = price

    def calculate_price(self):

        return self.__price


class OrderPriceDecorator(OrderPriceComponent):

    def __init__(self, component):

        self._component = component


class DiscountDecorator(OrderPriceDecorator):

    def __init__(self, component, discount_percentage):

        super().__init__(component)

        self.__discount_percentage = (discount_percentage)

    def calculate_price(self):

        price = (self._component.calculate_price() )

        discount = (price * self.__discount_percentage/ 100 )

        return price - discount


class ShippingDecorator(OrderPriceDecorator):

    def __init__(self, component, shipping_fee):

        super().__init__(component)

        self.__shipping_fee = shipping_fee

    def calculate_price(self):

        return (self._component.calculate_price() + self.__shipping_fee)



#Observe Pattern

class OrderObserver(ABC):

    @abstractmethod
    def update(self, order):
        pass


class EmailNotification(OrderObserver):

    def update(self, order):

        customer = order.get_customer()

        print( f"Email: Order "f"{order.get_order_id()} "f"confirmed for " f"{customer.get_name()}" )


class SMSNotification (OrderObserver):

    def update(self, order):

        customer = order.get_customer()

        print(f"SMS: Order "f"{order.get_order_id()} "f"confirmed for "f"{customer.get_name()}" )


class OrderNotifier:

    def __init__(self):

        self.__subscribers = set()

    def subscribe(self, subscriber):

        self.__subscribers.add(subscriber)

    def notify(self, order):

        for subscriber in self.__subscribers:

            subscriber.update(order)

class ModelRegistry:

    def __init__(self):

        self.__models = {}

    def register_model(self,model_name,version,model):

        self.__models[model_name] = {"version": version, "model": model}

class FeatureStore:
    def __init__ (self):
        self.__features={} 

    def set_feature(self,entity_id, feature_name,value):
        if entity_id not in self.__features:
            self.__features[entity_id]={}
        self.__features[entity_id][feature_name]=value 
                   
    def get_feature(self,entity_id,feature_name):
        entity_features=self.__features.get(entity_id,{})
        return entity_features.get(feature_name) 

    def get_all_features(self,entity_id):
        return self.__features.get(entity_id,{})

🟢 First, I wrote unit test using unittest.TestCase for two parts of ShopEase :authentication and order notification. I used setup() then I checked three cases; correct login,wrong password, and unknown user ID.

For TestOrderNotifications, I mocked OrderObserver with Mock(spec=OrderObserver) so I could test  OrderObserver  without sending  a real emails or SMS messages. the test fails as expected and shows that the essential code is missing. As Tibi(2022) mentions this in TDD “Failure is what we were looking for”.


For this capstone project I built ShopEase, a small online shope system. I used object Oriented design, Dependency Injection (DI), and some of design pattern. I used encapsulation throughout private attributes with getter methods  and abstraction class like Cache, User and PaymentStrategy class to make the payment methods changeable, DiscountDecorator to build up an order`s price step by step, Observer to notify multi[le listeners when an order is confirmed , Adapter to reuse an old LegacySystem, visitor to add new products operation without editing product and Abstract Factory(Web,Mobile) to create matching UI parts. Also I hashed the passwords with bcrypt rather than stored as a plain text. 

In addition, I used Dependecny Injection in some part of my code, for example, services like AuthenticationService and ProductService receive their dependencies through the constructor instead of creating them internally, making the code easier to test and change.

Finally, I added a FeatureStore to hold reusable data about users and RecommendationService that depends on a RecommendationModel interface rather than one fixed model that`s help to change the model later. 
I used unittest.mock.Mock to test classes in isolation, including AI dependent code without needing real databases or models.
