During this seminar, I learned how structural design patterns makes the software  flexible, reusable, and maintainable components.And how classes and objects are connected toghether.
I also practised applying the Decorator pattern using Python.
I developed  a coffee shop application. It includes Object-Oriented Programming, Decorator pattern  and SOLID principles. I implemented abstraction, inheritance and polymorphism.
The design follows the Open/Close Principle because a new ingredient can be added without modifying the existing  classes.

```Python
from abc import ABC , abstractmethod

#Abstraction

class coffee(ABC):
    @abstractmethod
    def coffee_type(self):
        pass

    @abstractmethod
    def coffee_cost(self):
        pass

class RegularCoffee (coffee):                   #Regular Coffee
    def coffee_type(self):
        return  "Regular Coffee"
    
    def coffee_cost(self):
        return 2.50

class CoffeeDecorator(coffee):                   #Decorator

    def __init__(self,coffee):
        self.coffee = coffee

    def coffee_type(self):
        return self.coffee.coffee_type()   

    def coffee_cost(self):
        return self.coffee.coffee_cost()

    
class Milk(CoffeeDecorator):                       # Concrate Decorator

    def coffee_type(self):
        return self.coffee.coffee_type() + ", Milk"

    def coffee_cost(self):
        return self.coffee.coffee_cost() + 1.00


class ChoclateDuster(CoffeeDecorator):
      def coffee_type(self):
             return self.coffee.coffee_type() + ", ChoclateDuster"

      def coffee_cost(self):
              return self.coffee.coffee_cost() + 0.10

  #Customise the Coffee
coffee = RegularCoffee()                                     

coffee = Milk(coffee)
coffee = ChoclateDuster(coffee)

#Dispaly Order
print("Order:",coffee.coffee_type())
print(f"Total:£{coffee.coffee_cost():.2f}")
```
